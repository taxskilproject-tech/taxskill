```powershell
supabase test db --local --workdir test-results/local/h11-lab C:/Proyectos/tax-skill/supabase/tests/h11-followup.sql C:/Proyectos/tax-skill/supabase/tests/h07-reporting.sql
yarn test
yarn lint
yarn check:architecture
yarn tsx scripts/run-h11-lab.ts seed
yarn tsx scripts/run-h11-lab.ts types
yarn tsx scripts/run-h11-lab.ts build
yarn tsx scripts/run-h11-lab.ts bundle
yarn tsx scripts/run-h11-lab.ts start
yarn tsx scripts/run-h11-lab.ts ui
```
